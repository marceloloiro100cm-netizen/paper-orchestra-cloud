\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage[%(babel)s]{babel}

\usepackage[a4paper,margin=2.54cm]{geometry}
\usepackage{setspace}
\doublespacing
% Recuo de 1,27 cm (0,5 polegada) em todos os paragrafos, inclusive o primeiro.
\usepackage{indentfirst}

\setlength{\parskip}{0pt}
% Numero da pagina no canto superior direito.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[R]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\fancypagestyle{plain}{\fancyhf{}\fancyhead[R]{\thepage}\renewcommand{\headrulewidth}{0pt}}
% Titulos APA sem numeracao: nivel 1 centralizado em negrito, nivel 2 a esquerda em negrito,

\usepackage{titlesec}
\setcounter{secnumdepth}{0}
\titleformat{\section}{\normalfont\normalsize\bfseries\filcenter}{}{0pt}{}
\titleformat{\subsection}{\normalfont\normalsize\bfseries}{}{0pt}{}
\titleformat{\subsubsection}{\normalfont\normalsize\bfseries\itshape}{}{0pt}{}
\titleformat{\paragraph}[runin]{\normalfont\normalsize\bfseries}{}{0pt}{}[.]
\titlespacing*{\section}{0pt}{\baselineskip}{0pt}
\titlespacing*{\subsection}{0pt}{\baselineskip}{0pt}
\titlespacing*{\subsubsection}{0pt}{\baselineskip}{0pt}
\titlespacing*{\paragraph}{1.27cm}{0pt}{0.5em}
% Resumo em pagina propria, com o rotulo centralizado em negrito e sem recuo.
\renewenvironment{abstract}{\clearpage\begin{center}\bfseries\abstractname\end{center}\noindent\ignorespaces}{\par\clearpage}
\usepackage{csquotes}
\usepackage[style=apa,backend=biber,natbib=true]{biblatex}
\addbibresource{references.bib}

\let\cite\parencite
\providecommand{\citeonline}{\textcite}
\usepackage[hidelinks]{hyperref}
